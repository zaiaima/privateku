## EmpMapper接口

```
package com.orcal.dao;



import java.util.List;



import org.apache.ibatis.annotations.Delete;

import org.apache.ibatis.annotations.Insert;

import org.apache.ibatis.annotations.Select;

import org.apache.ibatis.annotations.Update;



import com.oracel.entity.Emp;



public interface EmpMapper {



​	@Select("select * from emp where empno = #{empno}")

​	public Emp selectById(int empno);



​	@Select("select * from emp")

​	public List<Emp> selectAll();



​	@Delete("delete from emp where empno = #{empno}")

​	public int delete(int empno);



​	@Insert("insert into emp values(#{empno},#{ename},#{job},#{mgr},#{hiredate},#{sal},#{comm},#{deptno})")

​	public int add(Emp emp);



​	@Update("update emp set ename=#{ename},job=#{job},mgr=#{mgr},"

​			+ "hiredate=#{hiredate},sal=#{sal},comm=#{comm},deptno=#{deptno} "

​			+ "where empno=#{empno}")

​	public int update(Emp emp);



}
```

```
package com.orcal.dao;



import java.util.List;



import org.apache.ibatis.annotations.Insert;

import org.apache.ibatis.annotations.Select;

import org.apache.ibatis.annotations.Update;



import com.oracel.entity.Dept;



public interface DeptMapper {



​	@Select("select * from Dept where deptno = #{deptno}")

​	public Dept selectById(int deptno);

​	

​	@Select("select * from Dept")

​	public List<Dept> selectAll();

​	

​	@Insert("insert into dept values(#{deptno},#{dname},#{loc})")

​	public int add(Dept dept);

​	

​	@Update("update dept set dname=#{dname},loc=#{loc} where deptno=#{deptno}")

​	public int update(Dept dept);

}
```
