select * from country where name = 'China';
select * from city where Code = 'CHN';

select c.*,ct.Name from city c
    inner join country ct on ct.Code=c.CountryCode
    where c.CountryCode = 'CHN';
-- 查询出中国的所有城市信息、国家名称、官方用语
select * from countrylanguage where countrylanguage = 'CHN';

select c.*,ct.Name,cl.Language from city c
    inner join country ct on ct.Code=c.CountryCode
    inner join countrylanguage cl on cl.CountryCode=c.CountryCode
    where c.CountryCode ='CHN';
