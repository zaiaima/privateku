## 加法

```vue
<!DOCTYPE html>
<html>
	<head>
		<meta charset="utf-8" />
		<title></title>
		<!-- <script src="https://unpkg.com/vue@next"></script> -->
		<script src="js/v3.2.8/vue.global.prod.js" type="text/javascript" charset="utf-8"></script>
	</head>
	<body>
		<div id="app">
			<h1>course:{{ counter }}</h1>
			<h2>title：{{title}}</h2>
			<h4 v-bind:title="message">我是标题`</h4>
		</div>
		
		
		
		<script>
			const app = Vue.createApp({
						data() {
							return {
								counter: 0,
								title: 'i am title',
								message:'喵喵喵~'
							}
						},
					methods: {
						increment() {
							this.counter++;
							console.log('将counter加一');
						}
					}
			});
			const vm=app.mount('#app');
			
			// document.write(vm.counter);
			vm.increment();
		</script>
	</body>
</html>
```

## 条件查询

```vue
<body>
	<h3 v-if="type=='A'">分支一</h3>
	<h3 v-else-if="type=='B'">分之二</h3>
	<h3 v-else-if="type=='C'">分之三</h3>
	<h3 v-else>idon'tknow</h3>
	
	<script>
	const app = Vue.createApp({
				data() {
					return {
					counter: 0,
					title: 'i am title',
					message: 'hello',
					imgSrc: 'img/2.jfif',
					type: 'A',
					score: 100
				}
			}
		});
		const vm = app.mount('#app');
	</script>
</body>
```

## for循环

```vue
<!DOCTYPE html>
<html>
	<head>
		<meta charset="utf-8" />
		<title></title>
		<!-- <script src="https://unpkg.com/vue@next"></script> -->
		<script src="js/v3.2.8/vue.global.prod.js" type="text/javascript" charset="utf-8"></script>
	</head>
	<body>
		<div id="app">
			<ul>
				<li v-for="(item,index) in items">
					{{item.name}}---{{item.age}}
				</li>
			</ul>
			
			<ul>
				<li v-for="i in width">
					{{i}} 
				</li>
			</ul>
		</div>



		<script>
			const app = Vue.createApp({
				data() {
					return {
						items:[
							{name:'小久',age:18},
							{name:'小杨',age:180}
						],
						width:8,
					}
				}
			});
			const vm = app.mount('#app');

		</script>
	</body>
</html>

```
