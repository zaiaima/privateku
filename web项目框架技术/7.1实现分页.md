```java
package com.oracel.entity;

import java.util.List;

public class PaperInfo<T> {
//	当前页
	private int crtpaper;
//	每页的行数
	private int papercnt;
//	总行数
	private int totlacnt;
//	总页数
	private int totlapaper;
//	分页数据
	private List<T> paperdata;
	public int getCrtpaper() {
		return crtpaper;
	}
	public void setCrtpaper(int crtpaper) {
		this.crtpaper = crtpaper;
	}
	public int getPapercnt() {
		return papercnt;
	}
	public void setPapercnt(int papercnt) {
		this.papercnt = papercnt;
	}
	public int getTotlacnt() {
		return totlacnt;
	}
	public void setTotlacnt(int totlacnt) {
		this.totlacnt = totlacnt;
	}
	public int getTotlapaper() {
		return totlapaper;
	}
	public void setTotlapaper(int totlapaper) {
		this.totlapaper = totlapaper;
	}
	public List<T> getPaperdata() {
		return paperdata;
	}
	public void setPaperdata(List<T> paperdata) {
		this.paperdata = paperdata;
	}
	public PaperInfo(int crtpaper, int papercnt, int totlacnt, int totlapaper, List<T> paperdata) {
		super();
		this.crtpaper = crtpaper;
		this.papercnt = papercnt;
		this.totlacnt = totlacnt;
		this.totlapaper = totlapaper;
		this.paperdata = paperdata;
	}
	public PaperInfo() {
		this.crtpaper = 1;
		this.papercnt = 10;
	}
	
	
}
```

## EmpDaoImpl接口实现类

新增一个getpaper的实现

```java
@Override
	public PaperInfo<Emp> getpaper(PaperInfo<Emp> paperinfo) throws Exception {
		List<Emp> emplist = new ArrayList<Emp>();
		Connection conn = ConnUtil.getConn();
		PreparedStatement ps = conn.prepareStatement("select SQL_CALC_FOUND_ROWS * from emp limit ?,?");
		ps.setInt(1, (paperinfo.getCrtpaper()-1)*paperinfo.getPapercnt());
		ps.setInt(2, paperinfo.getPapercnt());
		ResultSet rs = ps.executeQuery();
		Emp emp = null;
		while (rs.next()) {
			emp = new Emp(rs.getInt("empno"), rs.getString("ename"), rs.getString("job")
				, rs.getInt("mgr"), rs.getDate("hiredate"), rs.getInt("sal"), rs.getInt("comm"), new Dept(rs.getInt("deptno")));
			emplist.add(emp);
		}
		paperinfo.setPaperdata(emplist);
		ps=conn.prepareStatement("select FOUND_ROWS()");
		rs=ps.executeQuery();
		if (rs.next()) {
			paperinfo.setTotlacnt(rs.getInt(1));
		}
		ConnUtil.close(ps, rs);
		return paperinfo;
	}
```

```java
	@Override
	public PaperInfo<Emp> getpaper(PaperInfo<Emp> paperinfo)  {
		try {
			paperinfo = empDao.getpaper(paperinfo);
			Dept dept = null;
			for (Emp e : paperinfo.getPaperdata()) {
				dept = deptDao.queryByNo(e.getDept().getDeptno());
				e.setDept(dept == null ? new Dept() : dept);
			}
		} catch (Exception e) {
			e.printStackTrace();
		}
		return paperinfo;
	}
```

```java
protected void doGet(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException {
		
		PaperInfo<Emp> paperinfo=new PaperInfo<>();
		String paperStr=request.getParameter("paper");
		if (paperStr != null && !"".equals(paperStr) ) {
			paperinfo.setCrtpaper(Integer.parseInt(paperStr));
		}
		
		paperinfo.setPapercnt(4);
		request.setAttribute("paperinfo", empService.getpaper(paperinfo));
		
		request.getRequestDispatcher("/List.jsp").forward(request, response);
	}
```
