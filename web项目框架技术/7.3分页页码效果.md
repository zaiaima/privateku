	function databind(data){
		var str = "";
		for (var i = 0; i < data.length; i++) {
			str += "<tr><td>"+data[i].empno+"</td><td>"+
			data[i].ename+"</td><td>"+data[i].job+"</td><td>"
			+data[i].sal+"</td><td>"+data[i].dept.dname+"</td></tr>";
		}
		$('#list tbody').html(str);

	};
	//页面加载
	getPage(1, true);
	//单机事件
    $('#btn').click(function() {
        getPage(1, false);
});
    //获取完整分页数据
	function getPage(page,isPaging){
		var param={};
		param.name=$("#txtName").val();
		param.job=$("#sltJob").val();
		param.page=page;
		$.ajax({
			type:"POST",
			url:"EmpSearchServlet",
			data:param,
			dataType:"json",			
			success:function(msg){
				console.log(msg)
				//绑定列表
				databind(msg.pageData);
				if(isPaging){
					$("#paging").paging({
				        pageNum: msg.crtPage, // 当前页面
				        totalNum: msg.totalPage, // 总页码
				        totalList: msg.totalCnt, // 记录总数量
				        callback: function (num) { //回调函数
				            console.log(num);
				            getPage(num, true);
				        }
				    });
					$('#paging a').click(function() {
                        getPage($(this).text(), false);
                });
				}
              	}
		});
	}
	});
</script>
