package com.oracel.entity;

public class Dept {
	
	private int deptno;
	private String dname;
	private String loc;
	
	public int getDeptno() {
		return deptno;
	}
	public void setDeptno(int deptno) {
		this.deptno = deptno;
	}
	public String getDname() {
		return dname;
	}
	public void setDname(String dname) {
		this.dname = dname;
	}
	public String getLoc() {
		return loc;
	}
	public void setLoc(String loc) {
		this.loc = loc;
	}
	public Dept(int deptno, String dname, String loc) {
		super();
		this.deptno = deptno;
		this.dname = dname;
		this.loc = loc;
	}
	public Dept(int deptno) {
		super();
		this.deptno = deptno;
	}
	public Dept() {
		// TODO Auto-generated constructor stub
	}
	@Override
	public String toString() {
		return "Dept [deptno=" + deptno + ", dname=" + dname + ", loc=" + loc + "]";
	}
	

}

## DeptMapper接口

```java
package com.orcal.dao;

import java.util.List;

import com.oracel.entity.Dept;
import com.oracel.entity.Emp;

public interface DeptMapper {

	public Dept selectById(int deptno);
	public List<Dept> selectAll();
	
	public int add(Dept dept);
	public int update(Dept dept);
}

```

## DeptMapper.xml页面

```xml
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE mapper
  PUBLIC "-//mybatis.org//DTD Mapper 3.0//EN"
  "http://mybatis.org/dtd/mybatis-3-mapper.dtd">
<mapper namespace="com.orcal.dao.DeptMapper">
	<select id="selectById" resultType="Dept">
		select * from Dept where
		deptno = #{deptno}
	</select>

	<select id="selectAll" resultType="Dept">
		select * from Dept
	</select>

	<insert id="add" parameterType="Dept">
		insert into dept
		values(#{deptno},#{dname},#{loc})
	</insert>
	<update id="update" parameterType="Dept">
		update dept set
		dname=#{dname},loc=#{loc}
		where deptno=#{deptno}
	</update>
</mapper>
```
