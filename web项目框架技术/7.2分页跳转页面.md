## EmpServiceImpl接口实现类

更改empsearch方法

```java
	@Override
	public PaperInfo<Emp> empsearch(PaperInfo<Emp> paperinfo,Emp emp) {
		try {
			paperinfo = empDao.quertbycondition(paperinfo, emp);
			Dept dept = null;
			for (Emp e : paperinfo.getPaperdata()) {
				dept = deptDao.queryByNo(e.getDept().getDeptno());
				e.setDept(dept == null ? new Dept() : dept);
			}
		} catch (Exception e) {
			e.printStackTrace();
		}
		return paperinfo;
	}
```

```java
protected void doPost(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException {
		PrintWriter out = response.getWriter();
		PaperInfo<Emp> paperinfo=new PaperInfo<>();
//		获取参数，
		String name=request.getParameter("name");
		String job=request.getParameter("job");
//		当前页
		String paperStr=request.getParameter("paper");
		if (paperStr != null && !"".equals(paperStr) ) {
			paperinfo.setCrtpaper(Integer.parseInt(paperStr));
		}
		
		Emp param=new Emp();
		param.setEname(name);
		param.setJob(job);
		
//		获取数据
		paperinfo = empService.empsearch(paperinfo, param);
		
//		获取数据
		
		//序列化
		ObjectMapper mapper = new ObjectMapper();
		String result = mapper.writeValueAsString(paperinfo);
		
		out.print(result);
		out.close();
	}
```
